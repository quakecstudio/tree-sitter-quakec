================================================================================
QC-9.5, QC-9.6: #undef and conditional blocks, at file level and in a body
================================================================================

float a;
#else
float b;
#endif
#ifndef X
#undef DEBUG
#endif
void() f = {
#ifdef DEBUG
  a = 1;
#endif
};

--------------------------------------------------------------------------------

(source_file
  (preproc_ifdef
    name: (identifier))
  (declaration
    type: (simple_type)
    declarator: (declarator
      name: (identifier)))
  (preproc_else)
  (declaration
    type: (simple_type)
    declarator: (declarator
      name: (identifier)))
  (preproc_endif)
  (preproc_ifdef
    name: (identifier))
  (preproc_undef
    name: (identifier))
  (preproc_endif)
  (declaration
    type: (function_type
      return_type: (simple_type)
      parameters: (parameter_list))
    declarator: (declarator
      name: (identifier)
      body: (function_body
        (preproc_ifdef
          name: (identifier))
        (expression_statement
          (assignment_expression
            left: (identifier)
            right: (float_literal)))
        (preproc_endif)))))
